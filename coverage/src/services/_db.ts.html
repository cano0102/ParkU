      let estado: Celda['estado'] = 'disponible';

        estado = 'mantenimiento';

        const r = seededRatio(idx);
        if (r < 0.65) estado = 'no_disponible';
        else if (r < 0.75) estado = 'reservada';

        id: `c${idx++}`,
        parqueaderoId: pq.id,
        numero: `${prefix}-${String(i + 1).padStart(3, '0')}`,
